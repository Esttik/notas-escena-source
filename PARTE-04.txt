    this.salt = salt;
    this.clock = clock;
  }
  async claim(requestId: string, phone: string, sessionId: string) {

    const now = this.clock();
    const fingerprint = createHmac("sha256", this.salt)
      .update(JSON.stringify([phone, sessionId]))
      .digest("hex");

      await client.query("BEGIN");
      // One row lock serializes claims and rate accounting across all instances.

        rows: [lock],
      } = await client.query("SELECT * FROM bridge_lock WHERE id=1 FOR UPDATE");
      if (!lock) throw new BridgeError("LEDGER_CONFIG", 503);
      await client.query("DELETE FROM bridge_requests WHERE created<$1", [
        now - 86400000,
      ]);

        Math.floor(now / 3600000),
      ]);

        rows: [old],
      } = await client.query(
        "SELECT fingerprint FROM bridge_requests WHERE id=$1",
        [requestId],
      );

        throw new BridgeError(
          old.fingerprint === fingerprint ? "DUPLICATE" : "PAYLOAD_MISMATCH",
          409,
        );

        rows: [rate],
      } = await client.query("SELECT count FROM bridge_limits WHERE id=$1", [
        Math.floor(now / 3600000),
      ]);
      if (Number(rate?.count ?? 0) >= 20)
        throw new BridgeError("RATE_LIMIT", 429);
      if (Number(lock.lease) >= now || Number(lock.cooldown) > now)
        throw new BridgeError("BUSY_OR_COOLDOWN", 409);
      await client.query(
        "INSERT INTO bridge_requests VALUES($1,$2,'pending',$3)",
        [requestId, fingerprint, now],
      );
      await client.query(
        "INSERT INTO bridge_limits VALUES($1,1) ON CONFLICT(id) DO UPDATE SET count=bridge_limits.count+1",
        [Math.floor(now / 3600000)],
      );
      await client.query(
        "UPDATE bridge_lock SET request_id=$1,lease=$2 WHERE id=1",
        [requestId, now + 90000],
      );
      await client.query("COMMIT");
    } catch (error) {
      await client.query("ROLLBACK");
      throw error;
    } finally {
      client.release();
    }
  }
  async finish(id: string, status: string, cooldownMs = 2000) {
    const client = await this.pool.connect();
    try {
      await client.query("BEGIN");
      await client.query("SELECT id FROM bridge_lock WHERE id=1 FOR UPDATE");
      await client.query("UPDATE bridge_requests SET status=$1 WHERE id=$2", [
        status,
        id,
      ]);
      await client.query(
        "UPDATE bridge_lock SET request_id=NULL,lease=0,cooldown=GREATEST(cooldown,$1) WHERE id=1 AND request_id=$2",
        [this.clock() + cooldownMs, id],
      );
      await client.query("COMMIT");
    } catch (error) {
      await client.query("ROLLBACK");
      throw error;
    } finally {
      client.release();
    }
  }

          else signal.addEventListener("abort", abort, { once: true });
        });

  return new BridgeError("TELEGRAM_LOGIN", 503);
}

function body(req: IncomingMessage): Promise<string> {
  return new Promise((resolve, reject) => {
    let size = 0;
    const chunks: Buffer[] = [];
    req.on("data", (chunk: Buffer) => {
      size += chunk.length;
      if (size > 2048) {
        reject(new BridgeError("BODY_LIMIT", 413));
        return;
      }
      chunks.push(chunk);
    });
    req.on("end", () => resolve(Buffer.concat(chunks).toString("utf8")));
    req.on("error", () => reject(new BridgeError("INPUT", 400)));
  });
}

export class Ledger {

  constructor(path: string, salt: string) {
    this.salt = salt;
    this.db = new DatabaseSync(path);
    this.db.exec(`PRAGMA busy_timeout=3000;
      CREATE TABLE IF NOT EXISTS requests (id TEXT PRIMARY KEY, fingerprint TEXT NOT NULL, status TEXT NOT NULL, created INTEGER NOT NULL);
      CREATE TABLE IF NOT EXISTS lock (id INTEGER PRIMARY KEY, request_id TEXT, lease INTEGER NOT NULL DEFAULT 0, cooldown INTEGER NOT NULL DEFAULT 0);
      CREATE TABLE IF NOT EXISTS limits (id INTEGER PRIMARY KEY, count INTEGER NOT NULL);
      INSERT OR IGNORE INTO lock (id) VALUES (1);`);
  }
  claim(requestId: string, phone: string, sessionId: string) {
    const now = Date.now();
    this.db.prepare("DELETE FROM requests WHERE created<?").run(now - 86400000);

      .update(JSON.stringify([phone, sessionId]))
      .digest("hex");

      .prepare("SELECT fingerprint FROM requests WHERE id=?")
      .get(requestId);
    if (old)
      throw new BridgeError(
        old.fingerprint === fingerprint ? "DUPLICATE" : "PAYLOAD_MISMATCH",
        409,
      );

    this.db.prepare("DELETE FROM limits WHERE id<?").run(window);
    const rate = this.db
      .prepare(
        "INSERT INTO limits VALUES (?,1) ON CONFLICT(id) DO UPDATE SET count=count+1 RETURNING count",
      )
      .get(window);
    if (Number(rate?.count) > 20) throw new BridgeError("RATE_LIMIT", 429);
    try {
      this.db
        .prepare("INSERT INTO requests VALUES (?,?,'pending',?)")
        .run(requestId, fingerprint, now);
    } catch {
      throw new BridgeError("DUPLICATE", 409);
    }
    const locked = this.db
      .prepare(
        "UPDATE lock SET request_id=?,lease=? WHERE id=1 AND lease<? AND cooldown<=? RETURNING id",
      )
      .get(requestId, now + 90000, now, now);
    if (!locked) {
      this.db
        .prepare("UPDATE requests SET status='not_sent' WHERE id=?")
        .run(requestId);
      throw new BridgeError("BUSY_OR_COOLDOWN", 409);
    }
  }
  finish(id: string, status: string, cooldownMs = 2000) {
    this.db.prepare("UPDATE requests SET status=? WHERE id=?").run(status, id);

      .prepare(

import { BridgeError, loginError } from "./errors.ts";

export class LoginChallenge {
  id = 0;
  kind: "code" | "password" = "code";
  length: number | undefined;
  delivery = "app";
  error = "";
  failures = 0;
  pending: {
    resolve: (value: string) => void;
    reject: (error: Error) => void;
  } | null = null;

  wait(kind: "code" | "password", length?: number, delivery = "app") {
    if (this.pending) throw new BridgeError("CHALLENGE_BUSY", 409);
    this.kind = kind;
    this.length =
      Number.isSafeInteger(length) && length! > 0 && length! <= 20
        ? length
        : undefined;
    this.delivery = ["app", "sms", "call"].includes(delivery)
      ? delivery
      : "other";
    this.id++;
    return new Promise<string>((resolve, reject) => {
      this.pending = { resolve, reject };
    });
  }

  submit(id: unknown, value: unknown) {
    if (!this.pending || id !== this.id)
      throw new BridgeError("STALE_CHALLENGE", 409);
    if (typeof value !== "string" || !value || value.length > 511)
      throw new BridgeError("INPUT", 400);
    if (this.kind === "code") {
      value = value.trim().replace(/[ -]/g, "");
      if (
        !/^\d+$/.test(value as string) ||
        (this.length && (value as string).length !== this.length)
      )
        throw new BridgeError("CODE_FORMAT", 400);
    }
    const pending = this.pending;
    this.pending = null;
    this.error = "";
    pending.resolve(value as string);
  }

  recover(error: unknown) {
    const safe = loginError(error);
    if (
      ![
        "PHONE_CODE_INVALID",
        "PHONE_CODE_EMPTY",
        "PASSWORD_HASH_INVALID",
      ].includes(safe.code) ||
      ++this.failures >= 3
    )
      throw safe;
    this.error = safe.code;
  }

  cancel() {
    const pending = this.pending;
    this.pending = null;
    pending?.reject(new BridgeError("LOGIN_TIMEOUT", 504));
  }
}

export async function hiddenInput(label: string): Promise<string> {

    throw new BridgeError("INTERACTIVE_TERMINAL_REQUIRED", 503);

  process.stdin.setRawMode(true);

      process.stdin.off("data", data);
      process.stdin.setRawMode(false);
      process.stdin.pause();
      process.stdout.write("\n");
    };
    const data = (chunk: Buffer) => {

          done();
          reject(new BridgeError("LOGIN_CANCELLED", 400));
          return;
        }

          done();
          resolve(value);
          return;
        }
        if (char === "\u007f" || char === "\b") value = value.slice(0, -1);
        else if (char >= " " && value.length < 512) value += char;
      }
    };
    process.stdin.on("data", data);
  });
}

import { createServer } from "node:http";
import { join } from "node:path";
import { chmodSync } from "node:fs";
import { makeClient, UserTransport } from "./telegram-client.ts";
import { stateDirectory, loadSession } from "./vault.ts";
import { Ledger } from "./ledger.ts";
import { BridgeEngine } from "./engine.ts";
import { BridgeError } from "./errors.ts";
import { createHandler } from "./http.ts";
async function main() {
  const key = process.env.BRIDGE_ENCRYPTION_KEY ?? "",
    secret = process.env.BRIDGE_BEARER_TOKEN ?? "";
  if (secret.length < 40) throw new BridgeError("BEARER_CONFIG", 503);
  const host = process.env.BRIDGE_HOST || "127.0.0.1",
    port = Number(process.env.PORT || 8788);
  if (!Number.isSafeInteger(port) || port < 1 || port > 65535)
    throw new BridgeError("PORT_CONFIG", 503);
  const client = makeClient(loadSession(key));
  const dbPath = join(stateDirectory(), "ledger.sqlite"),
    ledger = new Ledger(dbPath, secret);
  chmodSync(dbPath, 0o600);
  let engine: BridgeEngine;
  try {
    engine = new BridgeEngine(new UserTransport(client), ledger, {
      finalSuffix: process.env.TELEGRAM_FINAL_SUFFIX ?? "",
      noResultText: process.env.TELEGRAM_NO_RESULT_TEXT,
    });
    await client.connect();
  } catch (e) {
    ledger.close();
    await client.disconnect();
    throw e;
  }
  const server = createServer(createHandler(engine, secret));
  server.requestTimeout = 12000;
  server.headersTimeout = 5000;
  server.maxHeadersCount = 30;
  server.keepAliveTimeout = 5000;
  server.listen(port, host, () =>
    console.log(
      `Puente disponible en ${host}:${port}. No expongas este puerto sin HTTPS.`,
    ),
  );
  server.on("error", () => {
    console.error("BRIDGE_LISTEN");
    process.exitCode = 1;
    void client.disconnect();
    ledger.close();
  });
  const stop = () => {
    server.closeAllConnections();
    server.close(() => {
      ledger.close();
      void client.disconnect().finally(() => process.exit(0));
    });
  };
  process.once("SIGINT", stop);
  process.once("SIGTERM", stop);
}
main().catch((e) => {
  console.error(e instanceof BridgeError ? e.code : "BRIDGE_START");
  process.exitCode = 1;
});

import type { TelegramClient } from "teleproto";
import { randomUUID } from "node:crypto";
import { writeFileSync } from "node:fs";
import { join } from "node:path";
import { normalizePhone } from "../../lib/domain.ts";
import { saveSession, stateDirectory } from "./vault.ts";
import { BridgeError } from "./errors.ts";
export async function recordSession(client: TelegramClient, phone: string) {
  const me = await client.getMe();
  if (me.bot || !me.phone || normalizePhone("+" + me.phone) !== phone)
    throw new BridgeError("ACCOUNT_MISMATCH", 403);
  const session = client.session.save();
  if (typeof session !== "string" || !session)
    throw new BridgeError("SESSION_SAVE", 503);
  saveSession(session, process.env.BRIDGE_ENCRYPTION_KEY ?? "");
  const holder = {
    phone,
    consentRef: "owner-login:" + randomUUID(),
    ownershipVerifiedAt: new Date().toISOString(),
    expiresAt: new Date(Date.now() + 7 * 86400000).toISOString(),
    scope: "lookup",
  };
  writeFileSync(join(stateDirectory(), "holder.json"), JSON.stringify(holder), {
    mode: 0o600,
  });
}
