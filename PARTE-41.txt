        },
      };
    },

    service: new Service(new Store(adapter), {}, provider, timeout),
  };

function deferredProvider() {
  let calls = 0,
    resolve!: (value: QueryResult) => void;
  let input!: ProviderInput;
  const provider: Provider = {
    mode: "demo",
    lookup(i) {
      calls++;
      input = i;
      return new Promise((r) => {
        resolve = r;
      });
    },
  };
  return {
    provider,
    get calls() {
      return calls;
    },
    get input() {
      return input;
    },
    finish(found = true) {
      resolve({
        requestId: input.requestId,
        phone: input.phone,
        final: true,
        found,
        ...(found ? { country: "España", name: "Persona inventada" } : {}),
      });
    },
  };
}

function harness(updates: unknown[][] = [[reply()]]) {
  const calls: { method: string; body: Record<string, unknown> }[] = [];

    const method = String(url).split("/").at(-1)!;
    const body = JSON.parse(String(init?.body ?? "{}"));
    calls.push({ method, body });
    if (method === "getMe") return ok({ id: 12345, is_bot: true });
    if (method === "getWebhookInfo") return ok({ url: "" });
    if (method === "getChat")
      return ok({ id: 54321, type: "private", username: "TrueCalleRobot" });
    if (method === "sendMessage")
      return ok({ message_id: 40, chat: { id: 54321, type: "private" } });
    if (method === "getUpdates") return ok(updates.shift() ?? []);
    throw Error("unexpected method");
  };
  return { calls, fetcher };
}
