            if (f.size > 10000) {
              setHint("El archivo de texto debe ocupar menos de 10 KB.");
            } else {
              insert("\n" + (await f.text()));
            }
          }
          e.target.value = "";
        }}
      />
      <span className="sr-only" role="status" aria-live="polite">
        {props.status}
      </span>
    </section>
  );
});
